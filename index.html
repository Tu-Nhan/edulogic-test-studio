Nhập Danh Sách Học Sinh Hàng Loạt

Nhập danh sách theo định dạng CSV (Mã_HS, Họ_và_Tên, Lớp, Mật_Khẩu) mỗi dòng 1 học sinh:

Hướng dẫn Kết nối Google Sheet qua Apps Script

  1. Mở file Google Sheet của bạn trên Google Drive (Tạo 2 tab: DanhSachHocSinh và KetQuaBaiThi).
  2. Vào menu: Tiện ích mở rộng (Extensions) → Apps Script.
  3. Xóa toàn bộ mã cũ và dán đoạn mã bên dưới vào trình biên tập:
// ==========================================
// GOOGLE APPS SCRIPT: EDULOGIC TEST STUDIO
// ==========================================

// 1. HÀM GET: Tải danh sách Học sinh về WebApp
function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("DanhSachHocSinh") || ss.getSheets()[0];
  var data = sheet.getDataRange().getValues();
  var students = [];
  
  for (var i = 1; i < data.length; i++) {
    if (data[i][0]) {
      var passVal = String(data[i][3]).trim();
      var isFirstLoginVal = data[i][4];
      
      students.push({
        code: String(data[i][0]).trim(),
        name: String(data[i][1]).trim(),
        className: String(data[i][2]).trim(),
        pass: passVal,
        isFirstLogin: isFirstLoginVal === true || String(isFirstLoginVal).toLowerCase() === "true" || passVal === "123456"
      });
    }
  }
  
  return ContentService.createTextOutput(JSON.stringify(students))
    .setMimeType(ContentService.MimeType.JSON);
}

// 2. HÀM POST: Phân loại Ghi Điểm Thi hoặc Cập nhật Mật khẩu
function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var payload = JSON.parse(e.postData.contents);
    var action = payload.action;

    // --- TRƯỜNG HỢP A: ĐỔI/RESET MẬT KHẨU HỌC SINH ---
    if (action === "updateStudentPass") {
      var sheet = ss.getSheetByName("DanhSachHocSinh") || ss.getSheets()[0];
      var data = sheet.getDataRange().getValues();
      var targetCode = String(payload.studentCode || payload.code || payload.studentId).trim().toUpperCase();
      var newPass = String(payload.newPassword || payload.pass).trim();
      var isReset = payload.isReset === true;
      var updated = false;

      for (var i = 1; i < data.length; i++) {
        if (String(data[i][0]).trim().toUpperCase() === targetCode) {
          sheet.getRange(i + 1, 4).setValue(newPass);
          sheet.getRange(i + 1, 5).setValue(isReset ? true : false);
          updated = true;
          break;
        }
      }

      return ContentService.createTextOutput(JSON.stringify({
        status: updated ? "success" : "error", 
        message: updated ? "Cập nhật mật khẩu thành công" : "Không tìm thấy Mã HS"
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // --- TRƯỜNG HỢP B: LƯU KẾT QUẢ BÀI THI ---
    else if (action === "saveExamResult" || !action) {
      var sheetResults = ss.getSheetByName("KetQuaBaiThi");
      if (!sheetResults) {
        sheetResults = ss.insertSheet("KetQuaBaiThi");
        sheetResults.appendRow(["Thời gian nộp", "Mã HS", "Họ và Tên", "Lớp", "Tên Bài Thi", "Điểm Số", "Tỷ Lệ (%)", "Trạng Thái"]);
      }

      sheetResults.appendRow([
        payload.submitTime || payload.timestamp || new Date().toLocaleString("vi-VN"),
        payload.studentCode || payload.studentId || "",
        payload.studentName || "",
        payload.studentClass || "",
        payload.testTitle || "",
        (payload.score || 0) + " / " + (payload.totalPoints || 10),
        (payload.percentage || 0) + "%",
        payload.status || (payload.passed ? "ĐẠT" : "CHƯA ĐẠT")
      ]);

      return ContentService.createTextOutput(JSON.stringify({
        status: "success", 
        message: "Đã ghi kết quả bài thi thành công"
      })).setMimeType(ContentService.MimeType.JSON);
    }

    return ContentService.createTextOutput(JSON.stringify({status: "error", message: "Hành động không hợp lệ"}))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
